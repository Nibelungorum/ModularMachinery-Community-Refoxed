# MMCR MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 NeoForge 26.1.2 上跑通一个多方块合成引擎的最小可行版本——能定义机器结构（多向结构匹配）、注册配方（3 类 IO：item / fluid / energy）、tick 调度执行。

**Architecture:** 翻译路线——把 MMCE 1.12.2 的核心概念映射到 NeoForge 26.1.2 官方能力。机器 / 配方数据走 NeoForge `Recipe<?>` 体系；结构匹配自写（算法沿用 MMCE）；Capability 全用 `Capabilities.ItemHandler/FluidHandler/EnergyStorage`；KubeJS 通过 `RecipeSchema` Java API 程序化注册（无 JSON schema 文件）；每个功能配 unit test 或 GameTest。

**Tech Stack:**
- Minecraft 26.1.2 + NeoForge 26.1.2.84
- Java 25 toolchain
- DeferredRegister / RegisterCapabilitiesEvent / ModConfigSpec / MapCodec / CustomPacketPayload
- JUnit 5 + AssertJ（unit test）
- NeoForge `GameTest` 框架（gametest）
- KubeJS 26.1.2-8.0.4（可选绑定；缺它仍能跑）

```java
// src/main/java/cn/howxu/mmcr/api/machine/BlockPredicate.java
package cn.howxu.mmcr.api.machine;

import net.minecraft.tags.TagKey;
import net.minecraft.world.level.block.Block;
import net.minecraft.world.level.block.state.BlockState;

import java.util.List;

public sealed interface BlockPredicate {

    boolean matches(BlockState state);

    record Air() implements BlockPredicate {
        @Override public boolean matches(BlockState state) { return state.isAir(); }
    }

    record Any() implements BlockPredicate {
        @Override public boolean matches(BlockState state) { return true; }
    }

    record OfBlock(Block block) implements BlockPredicate {
        @Override public boolean matches(BlockState state) { return state.is(block); }
    }

    record OfBlockState(BlockState state) implements BlockPredicate {
        @Override public boolean matches(BlockState s) { return this.state == s; }
    }

    record OfTag(TagKey<Block> tag) implements BlockPredicate {
        @Override public boolean matches(BlockState state) { return state.is(tag); }
    }

    record AnyOf(List<BlockPredicate> children) implements BlockPredicate {
        @Override public boolean matches(BlockState state) {
            for (var child : children) if (child.matches(state)) return true;
            return false;
        }
    }
}
```

```java
// src/main/java/cn/howxu/mmcr/api/machine/BlockArray.java
package cn.howxu.mmcr.api.machine;

import net.minecraft.core.BlockPos;
import org.jetbrains.annotations.Nullable;

import java.util.Map;
import java.util.function.ToIntFunction;

public record BlockArray(Map<BlockPos, BlockPredicate> pattern) {

    public @Nullable BlockPredicate get(BlockPos pos) { return pattern.get(pos); }
    public boolean isEmpty() { return pattern.isEmpty(); }

    public int width()  { return extent(p -> p.getX()); }
    public int height() { return extent(p -> p.getY()); }
    public int length() { return extent(p -> p.getZ()); }

    private int extent(ToIntFunction<BlockPos> axis) {
        if (pattern.isEmpty()) return 0;
        int min = Integer.MAX_VALUE, max = Integer.MIN_VALUE;
        for (BlockPos p : pattern.keySet()) {
            int v = axis.applyAsInt(p);
            if (v < min) min = v;
            if (v > max) max = v;
        }
        return max - min + 1;
    }
}
```

```java
// src/main/java/cn/howxu/mmcr/api/machine/StructureMatcher.java
package cn.howxu.mmcr.api.machine;

import net.minecraft.core.BlockPos;
import net.minecraft.core.Direction;
import net.minecraft.world.level.Level;

public final class StructureMatcher {

    private StructureMatcher() {}

    /**
     * Check if {@code pattern} matches the world at {@code ctrlPos} looking at {@code facing}.
     * Pattern positions are rotated so +Z aligns with the controller's facing direction.
     */
    public static boolean matches(BlockArray pattern, Level level, BlockPos ctrlPos, Direction facing) {
        if (pattern.isEmpty()) return false;
        for (var entry : pattern.pattern().entrySet()) {
            BlockPos patternPos = entry.getKey();
            BlockPos worldPos = rotatePos(patternPos, facing).offset(ctrlPos);
            if (!entry.getValue().matches(level.getBlockState(worldPos))) return false;
        }
        return true;
    }

    private static BlockPos rotatePos(BlockPos pos, Direction facing) {
        return switch (facing) {
            case NORTH -> new BlockPos(-pos.getX(), pos.getY(), -pos.getZ());
            case SOUTH -> pos;
            case WEST -> new BlockPos(pos.getZ(), pos.getY(), -pos.getX());
            case EAST -> new BlockPos(-pos.getZ(), pos.getY(), pos.getX());
            default -> pos;
        };
    }
}
```

```java
package cn.howxu.mmcr.api.recipe;

import com.mojang.serialization.Codec;
import com.mojang.serialization.MapCodec;
import net.minecraft.world.item.crafting.Ingredient;
import net.neoforged.neoforge.fluids.crafting.FluidIngredient;

public sealed interface MachineIngredient {

    /** Codec with `"type"` discriminator for Item / Fluid / Energy. */
    Codec<MachineIngredient> CODEC = Codec.STRING.dispatch(
            MachineIngredient::type,
            type -> switch (type) {
                case "item"   -> ItemIngredient.CODEC.xmap(i -> (MachineIngredient) i, i -> (ItemIngredient) i);
                case "fluid"  -> FluidIngredientEntry.CODEC.xmap(f -> (MachineIngredient) f, f -> (FluidIngredientEntry) f);
                case "energy" -> EnergyIngredient.CODEC.xmap(e -> (MachineIngredient) e, e -> (EnergyIngredient) e);
                default -> throw new IllegalStateException("Unknown ingredient type: " + type);
            }
    );

    String type();

    record ItemIngredient(Ingredient item, int count) implements MachineIngredient {
        public static final MapCodec<ItemIngredient> CODEC = com.mojang.serialization.codecs.RecordCodecBuilder.mapCodec(
                instance -> instance.group(
                        Ingredient.CODEC.fieldOf("item").forGetter(ItemIngredient::item),
                        Codec.INT.fieldOf("count").forGetter(ItemIngredient::count)
                ).apply(instance, ItemIngredient::new));
        @Override public String type() { return "item"; }
    }

    record FluidIngredientEntry(net.neoforged.neoforge.fluids.crafting.FluidIngredient fluid, int amount)
            implements MachineIngredient {
        public static final MapCodec<FluidIngredientEntry> CODEC = com.mojang.serialization.codecs.RecordCodecBuilder.mapCodec(
                instance -> instance.group(
                        FluidIngredient.CODEC.fieldOf("fluid").forGetter(FluidIngredientEntry::fluid),
                        Codec.INT.fieldOf("amount").forGetter(FluidIngredientEntry::amount)
                ).apply(instance, FluidIngredientEntry::new));
        @Override public String type() { return "fluid"; }
    }

    record EnergyIngredient(int fePerTick) implements MachineIngredient {
        public static final MapCodec<EnergyIngredient> CODEC = Codec.INT.fieldOf("fe_per_tick")
                .xmap(EnergyIngredient::new, EnergyIngredient::fePerTick);
        @Override public String type() { return "energy"; }
    }
}
```

```java
package cn.howxu.mmcr.api.recipe;

import com.mojang.serialization.MapCodec;
import net.minecraft.network.RegistryFriendlyByteBuf;
import net.minecraft.network.codec.StreamCodec;
import net.minecraft.world.item.ItemStack;

public class MachineRecipeSerializer implements net.minecraft.world.item.crafting.RecipeSerializer<MachineRecipe> {

    @Override public MapCodec<MachineRecipe> codec() { return MachineRecipe.CODEC; }

    @Override public StreamCodec<RegistryFriendlyByteBuf, MachineRecipe> streamCodec() {
        return StreamCodec.of(this::write, this::read);
    }

    private void write(RegistryFriendlyByteBuf buf, MachineRecipe r) {
        buf.writeResourceLocation(r.machineId());
        buf.writeVarInt(r.tickTime());
        // ingredients 暂用最简 codec 序列化（首期）
        buf.writeJsonWithCodec(net.minecraft.world.item.crafting.Ingredient.CODEC.listOf(),
                r.inputs().stream().map(i -> {
                    if (i instanceof MachineIngredient.ItemIngredient ii) return ii.item();
                    return net.minecraft.world.item.crafting.Ingredient.EMPTY;
                }).toList());
        buf.writeCollection(r.outputs(), (b, s) -> ItemStack.OPTIONAL_STREAM_CODEC.encode(b, s));
    }

    private MachineRecipe read(RegistryFriendlyByteBuf buf) {
        var mid = buf.readResourceLocation();
        int tt = buf.readVarInt();
        // 简化读：仅读 outputs（inputs 由 Task 15 完整 IO 逻辑补充）
        buf.readJsonWithCodec(net.minecraft.world.item.crafting.Ingredient.CODEC.listOf());
        var outs = buf.readList(b -> ItemStack.OPTIONAL_STREAM_CODEC.decode(b));
        return new MachineRecipe(null, mid, tt, java.util.List.of(), outs);
    }
}
```

```java
import cn.howxu.mmcr.api.machine.Machine;
import net.minecraft.resources.ResourceLocation;

import java.util.*;
import java.util.stream.Collectors;

public final class RecipeRegistry {

    private static final Map<ResourceLocation, MachineRecipe> RECIPES = new LinkedHashMap<>();

    private RecipeRegistry() {}

    public static void register(MachineRecipe recipe) {
        if (recipe.id() == null) throw new IllegalArgumentException("Recipe id null");
        RECIPES.put(recipe.id(), recipe);
    }

    public static List<MachineRecipe> byMachine(Machine machine) {
        return RECIPES.values().stream()
                .filter(r -> r.machineId().equals(machine.registryName()))
```

```java
    public void serverTick() {
        if (level == null || level.isClientSide()) return;
        if (machine == null) return;

        boolean formed = cn.howxu.mmcr.api.machine.StructureMatcher.matches(
                machine.pattern(), level, getBlockPos(),
                getBlockState().getValue(MachineControllerBlock.FACING));
        if (formed != isFormed()) setFormed(formed);
        if (!formed) {
            if (activeRecipe != null) setActiveRecipe(null);
            return;
        }

        if (activeRecipe == null) tryStartNewRecipe();
        if (activeRecipe != null) tickActiveRecipe();
    }

    private void tryStartNewRecipe() {
        var recipes = cn.howxu.mmcr.api.recipe.RecipeRegistry.byMachine(machine);
        for (var r : recipes) {
            if (canAcceptInputs(r)) {
                setActiveRecipe(r);
                setTickCounter(0);
                return;
            }
        }
    }

    private boolean canAcceptInputs(MachineRecipe r) {
        for (var ing : r.inputs()) {
            switch (ing) {
                case MachineIngredient.ItemIngredient ii -> {
                    if (findAndCheckItemBus(ii) == null) return false;
                }
                case MachineIngredient.FluidIngredientEntry fi -> {
                    if (findAndCheckFluidHatch(fi) == null) return false;
                }
                case MachineIngredient.EnergyIngredient ei -> {
                    if (findAndCheckEnergyHatch(ei) == null) return false;
                }
            }
        }
        return true;
    }

    private ItemBusBlockEntity findAndCheckItemBus(MachineIngredient.ItemIngredient ii) {
        for (int dx = -1; dx <= 1; dx++) {
            for (int dz = -1; dz <= 1; dz++) {
                var pos = getBlockPos().offset(dx, 0, dz);
                if (level.getBlockEntity(pos) instanceof ItemBusBlockEntity be && be.ioType() == IOType.INPUT) {
                    var handler = be.getItemHandler(null);
                    int have = 0;
                    for (int s = 0; s < handler.getSlots(); s++) {
                        var stack = handler.getStackInSlot(s);
                        if (ii.item().test(stack)) have += stack.getCount();
                    }
                    if (have >= ii.count()) return be;
                }
            }
        }
        return null;
    }

    private FluidHatchBlockEntity findAndCheckFluidHatch(MachineIngredient.FluidIngredientEntry fi) {
        for (int dx = -1; dx <= 1; dx++) {
            for (int dz = -1; dz <= 1; dz++) {
                var pos = getBlockPos().offset(dx, 0, dz);
                if (level.getBlockEntity(pos) instanceof FluidHatchBlockEntity be && be.ioType() == IOType.INPUT) {
                    var h = be.getFluidHandler(null);
                    if (h.getFluidInTank(0).getAmount() >= fi.amount()) return be;
                }
            }
        }
        return null;
    }

    private EnergyHatchBlockEntity findAndCheckEnergyHatch(MachineIngredient.EnergyIngredient ei) {
        for (int dx = -1; dx <= 1; dx++) {
            for (int dz = -1; dz <= 1; dz++) {
                var pos = getBlockPos().offset(dx, 0, dz);
                if (level.getBlockEntity(pos) instanceof EnergyHatchBlockEntity be && be.ioType() == IOType.INPUT) {
                    var h = be.getEnergyStorage(null);
                    if (h.getEnergyStored() >= ei.fePerTick()) return be;
                }
            }
        }
        return null;
    }

    private void tickActiveRecipe() {
        var r = activeRecipe;
        if (r == null) return;
        int next = tickCounter + 1;
        if (next >= r.tickTime()) {
            consumeAndProduce(r);
            setActiveRecipe(null);
            setTickCounter(0);
        } else {
            setTickCounter(next);
        }
    }

    private void consumeAndProduce(MachineRecipe r) {
        for (var ing : r.inputs()) {
            switch (ing) {
                case MachineIngredient.ItemIngredient ii -> {
                    var bus = findAndCheckItemBus(ii);
                    if (bus == null) continue;
                    var handler = bus.getItemHandler(null);
                    int left = ii.count();
                    for (int s = 0; s < handler.getSlots() && left > 0; s++) {
                        var stack = handler.getStackInSlot(s);
                        if (ii.item().test(stack)) {
                            int take = Math.min(left, stack.getCount());
                            handler.extractItem(s, take, false);
                            left -= take;
                        }
                    }
                }
                case MachineIngredient.FluidIngredientEntry fi -> {
                    var hatch = findAndCheckFluidHatch(fi);
                    if (hatch == null) continue;
                    hatch.getFluidHandler(null).drain(fi.amount(), net.neoforged.neoforge.fluids.capability.IFluidHandler.FluidAction.EXECUTE);
                }
                case MachineIngredient.EnergyIngredient ei -> {
                    var eh = findAndCheckEnergyHatch(ei);
                    if (eh == null) continue;
                    eh.getEnergyStorage(null).extractEnergy(ei.fePerTick() * r.tickTime(), false);
                }
            }
        }
        for (var out : r.outputs()) {
            for (int dx = -1; dx <= 1; dx++) {
                for (int dz = -1; dz <= 1; dz++) {
                    var pos = getBlockPos().offset(dx, 0, dz);
                    if (level.getBlockEntity(pos) instanceof ItemBusBlockEntity be && be.ioType() == IOType.OUTPUT) {
                        var h = be.getItemHandler(null);
                        for (int s = 0; s < h.getSlots() && !out.isEmpty(); s++) {
                            var remainder = h.insertItem(s, out.copy(), false);
                            out.setCount(remainder.getCount());
                        }
                        if (out.isEmpty()) break;
                    }
                }
            }
        }
    }
```

```java
// src/main/java/cn/howxu/mmcr/kubejs/MachineBuilderJS.java
package cn.howxu.mmcr.kubejs;

import cn.howxu.mmcr.api.machine.BlockArray;
import cn.howxu.mmcr.api.machine.BlockPredicate;
import cn.howxu.mmcr.api.machine.DynamicMachine;
import cn.howxu.mmcr.api.machine.MachineRegistry;
import dev.latvian.mods.kubejs.registry.BuilderBase;
import net.minecraft.core.BlockPos;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.world.level.block.Block;
import net.minecraft.world.level.block.state.BlockState;

import java.util.HashMap;
import java.util.Map;

public class MachineBuilderJS extends BuilderBase<DynamicMachine> {

    public transient String localizedName = "Unknown Machine";
    public transient BlockArray pattern = new BlockArray(Map.of());

    public MachineBuilderJS(ResourceLocation id) {
        super(id);
    }

    public MachineBuilderJS localizedName(String name) {
        this.localizedName = name;
        return this;
    }

    public MachineBuilderJS pattern(String grid, Map<String, Object> keys) {
        Map<BlockPos, BlockPredicate> map = new HashMap<>();
        String[] rows = grid.trim().split("\\s+");
        for (int y = 0; y < rows.length; y++) {
            String row = rows[y];
            for (int x = 0; x < row.length(); x++) {
                char c = row.charAt(x);
                if (c == '_' || c == '.' || c == ' ') continue;
                Object obj = keys.get(String.valueOf(c));
                if (obj == null) continue;
                BlockPredicate pred = switch (obj) {
                    case Block b -> new BlockPredicate.OfBlock(b);
                    case BlockState s -> new BlockPredicate.OfBlockState(s);
                    case BlockPredicate p -> p;
                    default -> throw new IllegalArgumentException("Unknown pattern key: " + obj);
                };
                map.put(new BlockPos(x, y, 0), pred);
            }
        }
        this.pattern = new BlockArray(map);
        return this;
    }

    @Override
    public DynamicMachine createObject() {
        return new DynamicMachine(id, localizedName, pattern);
    }

    @Override
    public void registerObject() {
        MachineRegistry.register(createObject());
    }

    @Override
    public dev.latvian.mods.kubejs.registry.BuilderType<?> getBuilderType() {
        return null;
    }
}
```

```java
// src/main/java/cn/howxu/mmcr/kubejs/MachineRecipeBuilderJS.java
package cn.howxu.mmcr.kubejs;

import cn.howxu.mmcr.api.machine.Machine;
import cn.howxu.mmcr.api.machine.MachineRegistry;
import cn.howxu.mmcr.api.recipe.MachineIngredient;
import cn.howxu.mmcr.api.recipe.MachineRecipe;
import cn.howxu.mmcr.api.recipe.RecipeRegistry;
import dev.latvian.mods.kubejs.recipe.RecipeJS;
import net.minecraft.core.registries.BuiltInRegistries;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.item.crafting.Ingredient;

import java.util.ArrayList;
import java.util.List;

public class MachineRecipeBuilderJS extends RecipeJS {

    public Machine machine;
    public int tickTime = 40;
    public final List<MachineIngredient> inputs = new ArrayList<>();
    public final List<ItemStack> outputs = new ArrayList<>();
    public int energyPerTick = 0;

    public MachineRecipeBuilderJS machine(String id) {
        var machineId = ResourceLocation.parse(id);
        this.machine = MachineRegistry.getMachine(machineId);
        if (this.machine == null) throw new IllegalArgumentException("Machine not found: " + id);
        return this;
    }

    public MachineRecipeBuilderJS tickTime(int t) { this.tickTime = t; return this; }

    public MachineRecipeBuilderJS itemInput(String itemId, int count) {
        var item = BuiltInRegistries.ITEM.get(ResourceLocation.parse(itemId));
        inputs.add(new MachineIngredient.ItemIngredient(Ingredient.of(new ItemStack(item)), count));
        return this;
    }

    public MachineRecipeBuilderJS itemOutput(String itemId, int count) {
        var item = BuiltInRegistries.ITEM.get(ResourceLocation.parse(itemId));
        outputs.add(new ItemStack(item, count));
        return this;
    }

    public MachineRecipeBuilderJS energyPerTick(int fe) {
        this.energyPerTick = fe;
        return this;
    }

    @Override
    public void build() {
        if (machine == null) throw new IllegalStateException("machine() not called");
        if (energyPerTick > 0) inputs.add(new MachineIngredient.EnergyIngredient(energyPerTick));
        var recipe = new MachineRecipe(this.id, machine.registryName(), tickTime, inputs, outputs);
        RecipeRegistry.register(recipe);
    }
}
```
